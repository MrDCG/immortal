### 1.2 悬停效果实现

```vue
<!-- src/components/GodCard.vue -->
<template>
  <div
    class="god-card"
    :style="cardStyle"
    @mousemove="handleMouseMove"
    @mouseleave="handleMouseLeave"
    @mouseenter="isHovered = true"
  >
    <img :src="god.image" :alt="god.name" class="god-image" />

    <!-- 悬停时显示的表情切换 -->
    <Transition name="fade">
      <img
        v-if="isHovered"
        :src="happyImage"
        :alt="`${god.name} happy`"
        class="god-image god-image-hover"
      />
    </Transition>
  </div>
</template>

<script setup lang="ts">
import { ref, computed } from 'vue';

interface Props {
  god: God;
}

const props = defineProps<Props>();
const isHovered = ref(false);
const rotateX = ref(0);
const rotateY = ref(0);

const happyImage = computed(() =>
  props.god.image.replace('default', 'happy')
);

const cardStyle = computed(() => ({
  transform: `perspective(1000px) rotateX(${rotateX.value}deg) rotateY(${rotateY.value}deg)`,
  transition: isHovered.value ? 'transform 0.3s ease' : 'transform 0.5s ease',
}));

const handleMouseMove = (e: MouseEvent) => {
  const rect = (e.currentTarget as HTMLElement).getBoundingClientRect();
  const centerX = rect.left + rect.width / 2;
  const centerY = rect.top + rect.height / 2;
  const maxX = 100;
  const maxY = 100;

  rotateY.value = ((e.clientX - centerX) / maxX) * 10;
  rotateX.value = -((e.clientY - centerY) / maxY) * 10;
};

const handleMouseLeave = () => {
  isHovered.value = false;
  rotateX.value = 0;
  rotateY.value = 0;
};
</script>

<style scoped>
.god-card:hover {
  transform: scale(1.02);
}

.fade-enter-active,
.fade-leave-active {
  transition: opacity 0.3s ease;
}

.fade-enter-from,
.fade-leave-to {
  opacity: 0;
}
</style>
```

### 1.4 连续上香特殊效果

```typescript
// src/composables/useIncenseStreak.ts
export function useIncenseStreak() {
  const streak = ref(0);
  const lastIncenseTime = ref(0);
  let streakTimeout: number | undefined;

  const addIncense = () => {
    const now = Date.now();
    const timeSinceLast = now - lastIncenseTime.value;

    // 5秒内连续上香计为连击
    if (timeSinceLast < 5000) {
      streak.value++;
    } else {
      streak.value = 1;
    }

    lastIncenseTime.value = now;

    // 重置连击计时器
    if (streakTimeout) {
      clearTimeout(streakTimeout);
    }
    streakTimeout = window.setTimeout(() => {
      streak.value = 0;
    }, 5000);
  };

  return { streak, addIncense };
}
```

```typescript
// src/composables/useSound.ts
interface UseSoundOptions {
  volume?: number;
  loop?: boolean;
  autoplay?: boolean;
}

export function useSound(
  soundPath: string,
  options: UseSoundOptions = {}
) {
  const { volume = 0.5, loop = false, autoplay = false } = options;
  const audioRef = ref<HTMLAudioElement | null>(null);

  onMounted(() => {
    const audio = new Audio(soundPath);
    audio.volume = volume;
    audio.loop = loop;

    if (autoplay) {
      audio.play().catch(console.error);
    }

    audioRef.value = audio;
  });

  onUnmounted(() => {
    if (audioRef.value) {
      audioRef.value.pause();
      audioRef.value.src = '';
    }
  });

  const play = () => {
    audioRef.value?.play().catch(console.error);
  };

  const pause = () => {
    audioRef.value?.pause();
  };

  const stop = () => {
    if (audioRef.value) {
      audioRef.value.pause();
      audioRef.value.currentTime = 0;
    }
  };

  return { play, pause, stop };
}
```

# 每日财神/喜神方位查询功能

## 功能需求

- 根据当日天干自动计算财神方位
- 根据当日天干自动计算喜神方位
- 在页面左上角显示当日方位信息
- 使用传统历法算法（基于《玉匣记》）

## 方位计算原理

### 财神方位计算（《玉匣记》方法）

```
甲乙东北是财神，丙丁向在西南寻，
戊己正北坐方位，庚辛正东去安身，
壬癸原来正南坐，便是财神方位真。
```

| 天干 | 财神方位 |
|:---:|:---:|
| 甲、乙 | 东北方 |
| 丙、丁 | 西南方 |
| 戊、己 | 正北方 |
| 庚、辛 | 正东方 |
| 壬、癸 | 正南方 |

### 喜神方位计算

```
甲己在艮乙庚乾，丙辛坤位安喜神；
丁壬只在离宫坐，戊癸游来在巽宫。
```

| 天干 | 喜神方位 |
|:---:|:---:|
| 甲、己 | 东北方（艮位） |
| 乙、庚 | 西北方（乾位） |
| 丙、辛 | 西南方（坤位） |
| 丁、壬 | 正南方（离位） |
| 戊、癸 | 东南方（巽位） |

## 工具函数实现

```typescript
// src/utils/chineseCalendar.ts

type TianGan = '甲' | '乙' | '丙' | '丁' | '戊' | '己' | '庚' | '辛' | '壬' | '癸';
type Direction = '正北' | '东北' | '正东' | '东南' | '正南' | '西南' | '正西' | '西北';

export function getDayStem(date: Date = new Date()): TianGan {
  const baseDate = new Date(1900, 0, 1);
  const daysDiff = Math.floor((date.getTime() - baseDate.getTime()) / 86400000);
  const stems: TianGan[] = ['甲', '乙', '丙', '丁', '戊', '己', '庚', '辛', '壬', '癸'];
  const stemIndex = ((daysDiff % 10) + 10) % 10;
  return stems[stemIndex];
}

export function getCaishenDirection(date: Date = new Date()): Direction {
  const stem = getDayStem(date);
  const caishenMap: Record<TianGan, Direction> = {

}
```
