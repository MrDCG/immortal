### 2.2 组合式函数设计

```typescript
// src/composables/useGodSwitch.ts
export function useGodSwitch(totalCount: number) {
  const direction = ref(0);
  const currentIndex = ref(0);

  const next = () => {
    direction.value = 1;
    currentIndex.value = (currentIndex.value + 1) % totalCount;
  };

  const prev = () => {
    direction.value = -1;
    currentIndex.value = (currentIndex.value - 1 + totalCount) % totalCount;
  };

  const goTo = (index: number) => {
    direction.value = index > currentIndex.value ? 1 : -1;
    currentIndex.value = index;
  };

  // 键盘事件
  onMounted(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      if (e.key === 'ArrowLeft') prev();
      if (e.key === 'ArrowRight') next();
    };
    window.addEventListener('keydown', handleKeyDown);
    onUnmounted(() => window.removeEventListener('keydown', handleKeyDown));
  });

  return { currentIndex, direction, next, prev, goTo };
}
```

### 2.3 触摸滑动支持

```typescript
// src/composables/useSwipe.ts
interface SwipeHandlers {
  onSwipeLeft?: () => void;
  onSwipeRight?: () => void;
}

export function useSwipe(handlers: SwipeHandlers) {
  const touchStart = ref(0);
  const touchEnd = ref(0);
  const minSwipeDistance = 50;

  const onTouchStart = (e: TouchEvent) => {
    touchEnd.value = 0;
    touchStart.value = e.targetTouches[0].clientX;
  };

  const onTouchMove = (e: TouchEvent) => {
    touchEnd.value = e.targetTouches[0].clientX;
  };

  const onTouchEnd = () => {
    if (!touchStart.value || !touchEnd.value) return;

    const distance = touchStart.value - touchEnd.value;
    const isLeftSwipe = distance > minSwipeDistance;
    const isRightSwipe = distance < -minSwipeDistance;

    if (isLeftSwipe && handlers.onSwipeLeft) {
      handlers.onSwipeLeft();
    }
    if (isRightSwipe && handlers.onSwipeRight) {
      handlers.onSwipeRight();
    }
  };

  return { onTouchStart, onTouchMove, onTouchEnd };
}

// 或使用 VueUse 提供的 useSwipe
import { useSwipe } from '@vueuse/core';
```

### 3.4 烟雾粒子系统

```vue
<!-- src/components/Incense/SmokeParticles.vue -->
<template>
  <div class="smoke-container">
    <TransitionGroup name="smoke">
      <div
        v-for="particle in particles"
        :key="particle.id"
        class="smoke-particle"
        :style="smokeStyle(particle)"
      />
    </TransitionGroup>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted, onUnmounted } from 'vue';

interface SmokeParticle {
  id: number;
  x: number;
  y: number;
  opacity: number;
  scale: number;
  drift: number;
}

const particles = ref<SmokeParticle[]>([]);

let intervalId: number;

onMounted(() => {
  // 每100ms生成一个新粒子
  intervalId = window.setInterval(() => {
    const newParticle: SmokeParticle = {
      id: Date.now(),
      x: 10 + (Math.random() - 0.5) * 10,
      y: 0,
      opacity: 0.6,
      scale: 0.5,
      drift: (Math.random() - 0.5) * 2,
    };
    particles.value.push(newParticle);

    // 3秒后移除粒子
    setTimeout(() => {
      particles.value = particles.value.filter((p) => p.id !== newParticle.id);
    }, 3000);
  }, 100);
});

onUnmounted(() => {
  clearInterval(intervalId);
});

const smokeStyle = (particle: SmokeParticle) => ({
  '--drift': `${particle.drift * 50}px`,
  '--opacity': particle.opacity,
  '--scale': particle.scale,
});
</script>

<style scoped>
.smoke-particle {
  position: absolute;
  width: 20px;
  height: 20px;
  background: radial-gradient(circle, rgba(200,200,200,0.8) 0%, transparent 70%);
  border-radius: 50%;
  animation: smoke-rise 3s ease-out forwards;
}

@keyframes smoke-rise {
  0% {
    transform: translateY(0) scale(var(--scale));
    opacity: var(--opacity);
  }
  100% {
```

### 3.5 主组件实现

```vue
<!-- src/components/Incense.vue -->
<template>
  <div class="incense-container">
    <button
      @click="handleIncense"
      :disabled="isBurning"
      class="incense-btn"
    >
      <template v-if="isBurning">
        <div class="incense-burning">
          <IncenseStick :progress="progress" />
          <SmokeParticles />
        </div>
      </template>
      <template v-else>
        <IncenseIcon />
      </template>
    </button>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue';
import { useIncenseStore } from '@/stores/incense';
import IncenseStick from './IncenseStick.vue';
import SmokeParticles from './SmokeParticles.vue';

const incenseStore = useIncenseStore();
const emit = defineEmits<{
  complete: [];
}>();

const isBurning = ref(false);
const progress = ref(0);

const handleIncense = () => {
  if (isBurning.value) return;

  isBurning.value = true;
  progress.value = 0;
  incenseStore.addIncense(incenseStore.currentGodId);

  // 3秒燃烧动画
  const duration = 3000;
  const interval = 50;
  let elapsed = 0;

  const timer = setInterval(() => {
    elapsed += interval;
    progress.value = elapsed / duration;

    if (elapsed >= duration) {
      clearInterval(timer);
      isBurning.value = false;
      // 触发祈福语弹窗
      emit('complete');
    }
  }, interval);
};
</script>
```

## 六、状态管理完善

### 6.1 Store完善

```typescript
// src/stores/incense.ts
import { defineStore } from 'pinia';
import { ref, computed } from 'vue';
import type { God, IncenseRecord } from '@/types';

export const useIncenseStore = defineStore('incense', () => {
  // 状态
  const currentGodIndex = ref(0);
  const incenseRecords = ref<Record<string, IncenseRecord>>({});

  // 计算属性
  const currentGodId = computed(() => currentGodIndex.value);

  // 方法
  const setCurrentGodIndex = (index: number) => {
    currentGodIndex.value = index;
  };

  const nextGod = () => {
    currentGodIndex.value = (currentGodIndex.value + 1) % 6;
  };

  const prevGod = () => {
    currentGodIndex.value = (currentGodIndex.value - 1 + 6) % 6;
  };

  const addIncense = (godId: string) => {
    const today = new Date().toISOString().split('T')[0];
    const key = `${godId}_${today}`;
    incenseRecords.value = {
      ...incenseRecords.value,
      [key]: {
        godId,
        date: today,
        count: (incenseRecords.value[key]?.count || 0) + 1,
      },
    };
  };

  const getTodayIncenseCount = (godId: string): number => {
    const today = new Date().toISOString().split('T')[0];
    const key = `${godId}_${today}`;
    return incenseRecords.value[key]?.count || 0;
  };

  const getIncenseRecord = (godId: string, date: string): IncenseRecord | undefined => {
    const key = `${godId}_${date}`;
    return incenseRecords.value[key];
  };

  const getCurrentGod = (gods: God[]): God | null => {
    return gods[currentGodIndex.value] || null;
  };

  const reset = () => {
    currentGodIndex.value = 0;
    incenseRecords.value = {};
  };

  return {
    currentGodIndex,
    currentGodId,
    incenseRecords,
    setCurrentGodIndex,
    nextGod,
    prevGod,
    addIncense,
    getTodayIncenseCount,
    getIncenseRecord,
    getCurrentGod,
    reset,
  };
}, {
  persist: {
    key: 'incense-storage',
    version: 1,
  },
});
```
